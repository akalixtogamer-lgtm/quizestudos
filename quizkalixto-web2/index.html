11) Considere a classe abaixo. Digite o que será impresso em cada linha (para "nada/recusado", digite null):

class Estoque {
  #quantidade;
  constructor(quantidade) {
    this.#quantidade = Number(quantidade);
  }
  repor(qtd)   { this.#quantidade += Number(qtd); return this.#quantidade; }
  retirar(qtd) {
    if (this.#quantidade - Number(qtd) < 0) return null;
    this.#quantidade -= Number(qtd);
    return this.#quantidade;
  }
}

const e = new Estoque("8");
